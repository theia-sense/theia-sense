#Dropzone.jsx
import React from "react";

export default function Dropzone({ onFilesAdded }) {
  const onDrop = (e) => {
    e.preventDefault();
    onFilesAdded(e.dataTransfer.files);
  };
  const onDragOver = (e) => e.preventDefault();

  const onInputChange = (e) => {
    onFilesAdded(e.target.files);
  };

  return (
    <div
      onDrop={onDrop}
      onDragOver={onDragOver}
      style={{
        border: "2px dashed #ccc",
        padding: "20px",
        marginBottom: "20px",
        cursor: "pointer",
      }}
    >
      <p>Drag & drop images here, or click to select files</p>
      <input
        type="file"
        multiple
        accept="image/*"
        onChange={onInputChange}
        style={{ display: "none" }}
        id="fileInput"
      />
      <label htmlFor="fileInput" style={{ cursor: "pointer", color: "blue" }}>
        Select Images
      </label>
    </div>
  );
}





#UploadedImages.jx
import React from "react";

export default function UploadedImages({ filenames }) {
  if (!filenames || filenames.length === 0) return null;

  return (
    <div>
      <h3>Uploaded Images</h3>
      <div style={{ display: "flex", flexWrap: "wrap", gap: 10 }}>
        {filenames.map((filename) => (
          <img
            key={filename}
            src={`http://localhost:8000/images/${filename}`} // adjust URL as needed
            alt={filename}
            style={{ width: 100, height: 100, objectFit: "cover" }}
          />
        ))}
      </div>
    </div>
  );
}




#ImageUploader.jsx
import React, { useState, useCallback } from "react";
import Dropzone from "./Dropzone";
import UploadedImages from "./UploadedImages";

export default function ImageUploader() {
  const [selectedFiles, setSelectedFiles] = useState([]);
  const [resultImages, setResultImages] = useState([]);

  const onFilesAdded = useCallback((files) => {
    setSelectedFiles((prev) => [...prev, ...Array.from(files)]);
  }, []);

  const uploadFiles = async () => {
    if (selectedFiles.length === 0) return alert("No files selected.");

    const formData = new FormData();
    selectedFiles.forEach((file) => formData.append("images", file));

    try {
      const res = await fetch("http://localhost:8000/upload", {
        method: "POST",
        body: formData,
      });
      if (!res.ok) throw new Error("Upload failed");

      const data = await res.json();
      setResultImages(data.filenames || []);
    } catch (err) {
      console.error(err);
      alert("Upload failed.");
    }
  };

  return (
    <div>
      <Dropzone onFilesAdded={onFilesAdded} />
      <button onClick={uploadFiles} disabled={selectedFiles.length === 0}>
        Upload {selectedFiles.length} images
      </button>
      <UploadedImages filenames={resultImages} />
    </div>
  );
}

import React, { useState, useCallback } from "react";
import axios from "axios";
import Dropzone from "./Dropzone";
import UploadedImages from "./UploadedImages";

export default function ImageUploader() {
  const [selectedFiles, setSelectedFiles] = useState([]);
  const [resultImages, setResultImages] = useState([]);

  const onFilesAdded = useCallback((files) => {
    setSelectedFiles((prev) => [...prev, ...Array.from(files)]);
  }, []);

  const uploadFiles = async () => {
    if (selectedFiles.length === 0) return alert("No files selected.");

    const formData = new FormData();
    selectedFiles.forEach((file) => formData.append("images", file));

    try {
      const res = await axios.post("http://localhost:8000/upload", formData, {
        headers: {
          "Content-Type": "multipart/form-data",
        },
      });
      setResultImages(res.data.filenames || []);
    } catch (err) {
      console.error(err);
      alert("Upload failed.");
    }
  };

  return (
    <div>
      <Dropzone onFilesAdded={onFilesAdded} />
      <button onClick={uploadFiles} disabled={selectedFiles.length === 0}>
        Upload {selectedFiles.length} images
      </button>
      <UploadedImages filenames={resultImages} />
    </div>
  );
}
